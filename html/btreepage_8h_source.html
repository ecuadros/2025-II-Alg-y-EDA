75 return first;
76 while( first < last )
77 {
78 size_t mid = (first+last)/2;

82 // first = mid+1;

84 // last = mid;

449template <typename Trait>
450bt_ErrorCode CBTreePage<Trait>::Insert(const keyType& key, const ObjIDType ObjID){
451 size_t pos = binary_search(m_Keys, 0, m_KeyCount, key, m_Compare);
452 bt_ErrorCode error = bt_ok;
453
454// if( pos < m_KeyCount && (keyType)m_Keys[pos] == key && m_Unique)
455// return bt_duplicate; // this key is duplicate
456 if( pos < m_KeyCount && (!m_Compare(key, m_Keys[pos].key) && !m_Compare(m_Keys[pos].key, key)) && m_Unique)
457 return bt_duplicate;
458
459 if( !m_SubPages[pos] ){ // this is a leave
460 ::insert_at(m_Keys, ObjectInfo(key, ObjID), pos);
461 m_KeyCount++;
462 if( Overflow() )
463 return bt_overflow;
464 return bt_ok;
465 }
466 else{
467 // recursive insertion
468 error = m_SubPages[pos]->Insert(key, ObjID);
469 if( error == bt_overflow ){
470 if( !RedistributeWith1Brother(pos) )
471 SplitChild(pos);
472 if( Overflow() ) // Propagate overflow
473 return bt_overflow;
474 return bt_ok;
475 }
476 }
477 if( Overflow() ) // node overflow
478 return bt_overflow;
479 return bt_ok;
480}

505template <typename Trait>
506bool CBTreePage<Trait>::RedistributeWith1Brother(size_t &pos)
507{
508 if( m_SubPages[pos]->Underflow() )
509 {
510 size_t NumberOfKeyOnLeft = 0,
511 NumberOfKeyOnRight = 0;
512 // is this the first element or there are more elements on right brother
513 if( pos > 0 )
514 NumberOfKeyOnLeft = m_SubPages[pos-1]->NumberOfKeys();
515 if( pos < NumberOfKeys() )
516 NumberOfKeyOnRight = m_SubPages[pos+1]->NumberOfKeys();
517
518 if( NumberOfKeyOnLeft > NumberOfKeyOnRight )
519 if( m_SubPages[pos-1]->NumberOfKeys() > m_SubPages[pos-1]->MinNumberOfKeys() )
520 RedistributeL2R(pos-1); // bring elements from left brother
521 else
522 if( pos == NumberOfKeys() )
523 return (--pos, false);
524 else
525 return false;
526 else //NumberOfKeyOnLeft < NumberOfKeyOnRight )
527 if( m_SubPages[pos+1]->NumberOfKeys() > m_SubPages[pos+1]->MinNumberOfKeys() )
528 RedistributeR2L(pos+1); // bring elements from right brother
529 else
530 if( pos == 0 )
531 return (++pos, false);
532 else
533 return false;
534 }
535 else // it is due to overflow
536 {
537 size_t FreeCellsOnLeft = GetFreeCellsOnLeft(pos), // Free Cells On Left
538 fcor = GetFreeCellsOnRight(pos); // Free Cells On Right
539
540 if( !FreeCellsOnLeft && !fcor && m_SubPages[pos]->IsFull() )
541 return false;
542 if( FreeCellsOnLeft > fcor ) // There is more space on left
543 RedistributeR2L(pos);
544 else
545 RedistributeL2R(pos);
546
547 }
548 return true;
549}

585template <typename Trait>
586bool CBTreePage<Trait>::RedistributeWith2Brothers(size_t pos)
587{
588 assert( pos > 0 && pos < NumberOfKeys() );
589 assert( m_SubPages[pos-1] != 0 && m_SubPages[pos] != 0 && m_SubPages[pos+1] != 0 );
590 assert( m_SubPages[pos-1]->Underflow() ||
591 m_SubPages[ pos ]->Underflow() ||
592 m_SubPages[pos+1]->Underflow() );
593
594 if( m_SubPages[pos-1]->Underflow() )
595 { // Rotate R2L
596 RedistributeR2L(pos+1);
597 RedistributeR2L(pos);
598 if( m_SubPages[pos-1]->Underflow() )
599 return false;
600 }
601 else if( m_SubPages[pos+1]->Underflow() )
602 { // Rotate L2R
603 RedistributeL2R(pos-1);
604 RedistributeL2R(pos);
605 if( m_SubPages[pos+1]->Underflow() )
606 return false;
607 }
608 else // The problem is exactly at pos !
609 {
610 // Rotate L2R
611 RedistributeL2R(pos-1);
612 RedistributeR2L(pos+1);
613 if( m_SubPages[pos]->Underflow() )
614 return false;
615 }
616 return true;
617}

638template <typename Trait>
639void CBTreePage<Trait>::RedistributeR2L(size_t pos)
640{
641 BTPage *pSource = m_SubPages[ pos ],
642 *pTarget = m_SubPages[pos-1];
643
644 while(pSource->GetNumberOfKeys() > pSource->MinNumberOfKeys() &&
645 pTarget->GetNumberOfKeys() < pSource->GetNumberOfKeys() )
646 {
647 // Move from this page to the down-left page \/
648 ::insert_at(pTarget->m_Keys, m_Keys[pos-1], pTarget->NumberOfKeys()++);
649 // Move the pointer leftest pointer to the rightest position
650 ::insert_at(pTarget->m_SubPages, pSource->m_SubPages[0], pTarget->NumberOfKeys());
651
652 // Move the leftest element to the root
653 m_Keys[pos-1] = pSource->m_Keys[0];
654
655 // Remove the leftest element from rigth page
656 ::remove(pSource->m_Keys , 0);
657 ::remove(pSource->m_SubPages, 0);
658 pSource->NumberOfKeys()--;
659 }
660}

725template <typename Trait>
726void CBTreePage<Trait>::SplitChild(size_t pos)
727{
728 // FIRST: deciding the second page to split
729 BTPage *pChild1 = 0, *pChild2 = 0;
730 if( pos > 0 ) // is left page full ?
731 if( m_SubPages[pos-1]->IsFull() )
732 {
733 pChild1 = m_SubPages[pos-1];
734 pChild2 = m_SubPages[pos--];
735 }
736 if( pos < GetNumberOfKeys() ) // is right page full ?
737 if( m_SubPages[pos+1]->IsFull() )
738 {
739 pChild1 = m_SubPages[pos];
740 pChild2 = m_SubPages[pos+1];
741 }
742 size_t nKeys = pChild1->GetNumberOfKeys() + pChild2->GetNumberOfKeys() + 1;
743
744 // SECOND: copy both pages to a temporal one
745 // Create two tmp vector
746 vector<ObjectInfo> tmpKeys;
747 //tmpKeys.resize(nKeys);
748 vector<BTPage *> tmpSubPages;
749 //tmpKeys.resize(nKeys+1);
750
751 // copy from left child
752 MovePage(pChild1, tmpKeys, tmpSubPages);
753 // copy a key from parent
754 tmpKeys .push_back(m_Keys[pos]);
755
756 // copy from right child
757 MovePage(pChild2, tmpKeys, tmpSubPages);
758
759 BTPage *pChild3 = 0;
760 ObjectInfo oi1, oi2;
761 SplitPageInto3(tmpKeys, tmpSubPages, pChild1, pChild2, pChild3, oi1, oi2);
762
763 // copy the first element to the root
764 m_Keys [pos] = oi1;
765 m_SubPages[pos] = pChild1;
766
767 // copy the second element to the root
768 ::insert_at(m_Keys, oi2, pos+1);
769 ::insert_at(m_SubPages, pChild2, pos+1);
770 NumberOfKeys()++;
771
772 m_SubPages[pos+2] = pChild3;
773}

970/*brief Aplica una función a cada clave en la página del árbol B, recorriéndola en orden.
971 *
972 * Este método recorre todas las claves almacenadas en la página del árbol B y aplica una función proporcionada a cada clave.
973 * La función también recibe el nivel actual en el árbol como argumento, lo que permite realizar operaciones específicas según la profundidad.
974 *
975 * @tparam Trait El tipo de los traits que define las propiedades del árbol B.
976 * @tparam Function El tipo de la función que se aplicará a cada clave. Debe ser invocable con dos argumentos: la clave y el nivel.
977 *
978 * @param fn La función que se aplicará a cada clave. Debe aceptar dos parámetros: la clave y el nivel.
979 * @param level El nivel actual en el árbol B, utilizado para proporcionar contexto adicional a la función aplicada.
980 *
981 * @details El método recorre las claves en orden ascendente. Para cada clave, primero se llama recursivamente a `ForEach` en la subpágina correspondiente (si existe),
982 * luego se aplica la función `fn` a la clave actual junto con el nivel, y finalmente se continúa con la siguiente clave. Al final, también se llama a `ForEach`
983 * en la última subpágina (si existe).
984 */
985template <typename Trait>
986template <typename Function>
987void CBTreePage<Trait>::ForEach(Function fn, size_t level){
988 for(size_t i = 0 ; i < m_KeyCount ; i++)
989 {
990 if( m_SubPages[i] )
991 m_SubPages[i]->ForEach(fn, level+1);
992
993 std::invoke(fn, m_Keys[i], level);
994 }
995 if( m_SubPages[m_KeyCount] )
996 m_SubPages[m_KeyCount]->ForEach(fn, level+1);
997}
