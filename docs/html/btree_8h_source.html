178 m_stack.push_back({n, 0});
179 n = n->m_SubPages[0];
180 }
181 if (!n || n->m_KeyCount == 0) {
182 // árbol vacío
183 m_curr = nullptr; m_idx = 0; m_stack.clear();
184 return;
185 }
186 m_curr = n; m_idx = 0;
187 }

189 void next() {
190 if (!m_curr) return; // ya en end
191 //si hay hijo derecho del elemento actual, bajamos a su mínimo
192 if (m_curr->m_SubPages[m_idx + 1]) {
193 BTNode* n = m_curr->m_SubPages[m_idx + 1];
194 m_stack.push_back({m_curr, m_idx + 1}); // subiremos desde aquí
195 while (n->m_SubPages[0]) {
196 m_stack.push_back({n, 0});
197 n = n->m_SubPages[0];
198 }
199 m_curr = n; m_idx = 0;
200 return;
201 }
202 // subir hasta poder avanzar en el mismo nodo
203 while (!m_stack.empty()) {
204 auto fr = m_stack.back(); m_stack.pop_back();
205 BTNode* parent = fr.node;
206 size_t pi = fr.idx;
207 // venimos de "antes" del key de índice pi; ese key es el siguiente
208 if (pi < parent->m_KeyCount) {

211 }
212 // si pi == m_KeyCount, estábamos después del último key; seguir subiendo
213 }

374size_t BTree<Trait>::computeHeight(const BTNode& n) const {
375 // hoja: altura 1
376 if (!n.m_SubPages[0]) return 1;
377 size_t best = 0;
378 for (size_t i = 0; i <= n.m_KeyCount; ++i) {
379 if (n.m_SubPages[i]) {
380 best = std::max(best, computeHeight(*n.m_SubPages[i]));
381 }
382 }
383 return best + 1;
384}

409template <typename Trait>
410bool BTree<Trait>::Load(const std::string& filename) {
411 std::unique_lock<std::shared_mutex> lk(m_mtx);
412 std::ifstream ifs(filename);
413 if (!ifs) return false;
414
415 size_t order = 0; int uniq = 1;
416 ifs >> order >> uniq;
417
418 // Si cambió el “order”, recrea root con la nueva capacidad de hijos
419 m_Order = order ? order : m_Order;
420 m_Unique = (uniq != 0);
421
422 // Reconfigura root para hijos
423 m_Root.SetMaxKeysForChilds(m_Order);
424
425 // Limpia y carga
426 m_Root.Reset();
427 m_Root.Read(ifs);
428
429 // Recalcula métricas
430 m_Height = computeHeight(m_Root);
431 m_NumKeys = computeSize(m_Root);
432
433 return true;
434}
